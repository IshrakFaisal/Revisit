cmake_minimum_required(VERSION 3.20)
project(revisit_context_proof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(protocol protocol.cpp)
target_include_directories(protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(protocol_test test/protocol_test.cpp)
target_link_libraries(protocol_test PRIVATE protocol)
add_test(NAME protocol COMMAND protocol_test)
add_executable(worker_test test/worker_test.cpp)
add_test(NAME bounded_worker COMMAND worker_test)
if(MSVC)
  target_compile_options(protocol PRIVATE /W4 /WX)
  target_compile_options(protocol_test PRIVATE /W4 /WX)
endif()
if(WIN32)
  add_executable(notification_settings_test test/notification_settings_test.cpp)
  target_compile_definitions(notification_settings_test PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(notification_settings_test PRIVATE windowsapp)
  add_test(NAME notification_settings COMMAND notification_settings_test)
  add_executable(activation_test test/activation_test.cpp)
  target_compile_definitions(activation_test PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(activation_test PRIVATE ole32)
  add_test(NAME activation_boundary COMMAND activation_test)
  add_library(pipe_transport pipe_transport.cpp)
  target_link_libraries(pipe_transport PUBLIC protocol advapi32)
  target_compile_definitions(pipe_transport PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_library(peer_identity peer_identity.cpp)
  target_link_libraries(peer_identity PUBLIC pipe_transport wintrust crypt32)
  add_executable(revisit_desktop_host desktop_host.cpp)
  target_link_libraries(revisit_desktop_host PRIVATE peer_identity)
  add_executable(peer_test test/peer_test.cpp)
  target_link_libraries(peer_test PRIVATE peer_identity)
  add_test(NAME peer_identity COMMAND peer_test)
  add_executable(revisit_host host.cpp)
  target_link_libraries(revisit_host PRIVATE pipe_transport)
  add_executable(revisit_receiver receiver.cpp)
  target_link_libraries(revisit_receiver PRIVATE pipe_transport)
  if(MSVC)
    foreach(target pipe_transport revisit_host revisit_receiver)
      target_compile_options(${target} PRIVATE /W4 /WX)
    endforeach()
  endif()
endif()
